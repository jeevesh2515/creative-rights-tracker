// server/lib/logging.js — Centralized logging

const fs = require('fs');
const path = require('path');

const LOG_DIR = process.env.LOG_DIR || path.join(__dirname, '../../logs');
const LOG_LEVEL = process.env.LOG_LEVEL || 'info';

// Ensure log directory exists
if (!fs.existsSync(LOG_DIR)) {
  fs.mkdirSync(LOG_DIR, { recursive: true });
}

const logger = {
  info: (message, data) => {
    const timestamp = new Date().toISOString();
    const logEntry = JSON.stringify({ timestamp, level: 'INFO', message, data });
    console.log(logEntry);
    appendLog('info.log', logEntry);
  },

  warn: (message, data) => {
    const timestamp = new Date().toISOString();
    const logEntry = JSON.stringify({ timestamp, level: 'WARN', message, data });
    console.warn(logEntry);
    appendLog('warn.log', logEntry);
  },

  error: (message, error, data) => {
    const timestamp = new Date().toISOString();
    const logEntry = JSON.stringify({
      timestamp,
      level: 'ERROR',
      message,
      error: error?.message || String(error),
      stack: error?.stack,
      data
    });
    console.error(logEntry);
    appendLog('error.log', logEntry);
  },

  debug: (message, data) => {
    if (LOG_LEVEL === 'debug') {
      const timestamp = new Date().toISOString();
      const logEntry = JSON.stringify({ timestamp, level: 'DEBUG', message, data });
      console.log(logEntry);
      appendLog('debug.log', logEntry);
    }
  }
};

function appendLog(filename, logEntry) {
  const logPath = path.join(LOG_DIR, filename);
  fs.appendFileSync(logPath, logEntry + '\n', { encoding: 'utf8' });
}

module.exports = logger;
```

const MAX_LOG_SIZE = 10 * 1024 * 1024; // 10MB

function appendLog(filename, logEntry) {
  const logPath = path.join(LOG_DIR, filename);
  
  // Check file size
  if (fs.existsSync(logPath)) {
    const stats = fs.statSync(logPath);
    if (stats.size > MAX_LOG_SIZE) {
      const timestamp = Date.now();
      fs.renameSync(logPath, logPath.replace('.log', `.${timestamp}.log`));
    }
  }
  
  fs.appendFileSync(logPath, logEntry + '\n', { encoding: 'utf8' });
}
```

## Integration Patterns

### Pattern 1: Allocating Rights (Admin Workflow)

1. Admin calls RevenueRights.allocateRights(userAddress, percentage)
2. Contract emits AllocationCreated event
3. Server listener captures event
4. Server writes to Supabase allocations table
5. Frontend listens to Supabase REALTIME and updates UI

### Pattern 2: Distributing Revenue (Admin Workflow)

1. Admin queries allocated rights from RevenueRights
2. Calculates amounts based on percentages and total revenue
3. Calls RevenueSplitter.split(recipients, amounts)
4. Contract emits DistributionCompleted + individual Transfer events
5. Server listener processes events and updates transaction table
6. Frontend shows distribution in transaction history

### Pattern 3: Querying User Balance (User Workflow)

1. User logs in (authenticated via Supabase)
2. Frontend calls getBalance(userAddress)
3. Contract returns balance from on-chain state
4. Frontend also queries transactions from Supabase for history

## Error Handling

### Common Errors

**"Unauthorized" / "AccessControl"** — Caller not authorized for operation
- Solution: Check caller has correct role (admin vs user)

**"Insufficient Balance"** — Trying to distribute more than available
- Solution: Check contract balance before distribution

**"Invalid Parameters"** — Array length mismatch or invalid address
- Solution: Validate inputs before calling

### Error Recovery

Contract errors are soft failures (revert with message). Always:
1. Wrap calls in try/catch
2. Log the error
3. Return user-friendly message
4. Retry with exponential backoff for transient failures

Example:
\`\`\`javascript
try {
  const tx = await contract.allocateRights(addr, pct);
  const receipt = await tx.wait();
} catch (error) {
  if (error.reason === 'Unauthorized') {
    // Handle auth error specifically
  } else {
    // Generic error
    logger.error('Contract call failed', error);
  }
}
\`\`\`

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Contract ↔ Database | Smart contract state vs Supabase database consistency |
| User ↔ Contract | User transactions on contract (unverified caller) |
| Admin ↔ Distribution | Admin permission checks before distribution |
| Listener ↔ Event | Event listener receiving unverified blockchain data |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-C1 | Spoofing | Contract address | Mitigate | Verify contract address in ENV var, hardcode checksums in code review |
| T-01-C2 | Tampering | Smart contract bytecode | Accept | Contracts deployed to blockchain (immutable), only modification risk is compromised private key |
| T-01-C3 | Repudiation | Transaction signatures | Mitigate | All transactions signed with private key, wallet controls key security |
| T-01-C4 | Info Disclosure | Event listener logs | Mitigate | Ensure error logs don't contain private keys, deploy with LOG_LEVEL=info (not debug in prod) |
| T-01-C5 | Denial of Service | Event listener | Mitigate | Listener handles reconnections, implements backoff, logging for monitoring |
| T-01-C6 | Elevation of Privilege | Role-based access | Mitigate | Contract access controls (onlyAdmin, onlyOwner), verify in tests |
| T-01-DB1 | Integrity | Database state sync | Mitigate | Reconciliation logic, audit logs, idempotent event handlers |
| T-01-DB2 | Confidentiality | User allocations in DB | Mitigate | Database encryption at rest, role-based queries (users see only own data), phase 6 security audit |
