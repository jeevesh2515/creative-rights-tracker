```typescript
'use client';
import React, { useState } from 'react';
import { Pie } from 'react-chartjs-2';
import { Chart as ChartJS, ArcElement, Tooltip, Legend } from 'chart.js';
import { formatCurrency } from '@/lib/utils';

ChartJS.register(ArcElement, Tooltip, Legend);

interface Revenue {
  projectId: string;
  projectName: string;
  amount: number;
  [key: string]: any;
}

export const ProjectEarningsBreakdown: React.FC = () => {
  const [data, setData] = useState<Revenue[]>([]);
  const [isLoading, setIsLoading] = useState(true);

  React.useEffect(() => {
    const fetch = async () => {
      try {
        const response = await fetch('/api/revenue?ts=' + Date.now());
        if (!response.ok) throw new Error();
        const arr = await response.json();
        setData(Array.isArray(arr) ? arr : []);
      } catch {
        setData([]);
      }
      setIsLoading(false);
    };
    fetch();
  }, []);

  // Aggregate by project
  const aggregated = React.useMemo(() => {
    const map = new Map<string, { name: string; total: number }>();
    for (const item of data) {
      const key = item.projectId;
      if (!map.has(key)) {
        map.set(key, { name: item.projectName, total: 0 });
      }
      map.get(key)!.total += item.amount;
    }
    return Array.from(map.entries()).map(([id, { name, total }]) => ({ id, name, total }));
  }, [data]);

  const chartData = React.useMemo(() => {
    const colors = [
      '#3b82f6', '#ef4444', '#10b981', '#f59e0b', '#8b5cf6',
      '#ec4899', '#14b8a6', '#f97316', '#6366f1', '#06b6d4',
    ];

    return {
      labels: aggregated.map(p => p.name),
      datasets: [{
        label: 'Earnings by Project',
        data: aggregated.map(p => p.total),
        backgroundColor: aggregated.map((_, i) => colors[i % colors.length]),
        borderColor: 'rgba(255, 255, 255, 0.8)',
        borderWidth: 2,
      }],
    };
  }, [aggregated]);

  const chartOptions = {
    responsive: true,
    maintainAspectRatio: true,
    plugins: {
      legend: {
        display: true,
        position: 'right' as const,
      },
      tooltip: {
        callbacks: {
          label: (context: any) => {
            const total = context.dataset.data.reduce((a: number, b: number) => a + b, 0);
            const value = context.parsed ?? 0;
            const percentage = ((value / total) * 100).toFixed(1);
            return `${formatCurrency(value)} (${percentage}%)`;
          },
        },
      },
    },
  };

  if (isLoading) {
    return (
      <div className="col-span-1 md:col-span-2 lg:col-span-1">
        <div className="h-80 bg-gray-100 animate-pulse rounded-lg" />
      </div>
    );
  }

  if (aggregated.length === 0) {
    return (
      <div className="col-span-1 md:col-span-2 lg:col-span-1">
        <div className="h-80 bg-white rounded-lg border border-gray-100 flex items-center justify-center">
          <div className="text-gray-500">No project earnings data</div>
        </div>
      </div>
    );
  }

  return (
    <div className="col-span-1 md:col-span-2 lg:col-span-1 bg-white rounded-lg border border-gray-100 p-6">
      <h3 className="text-lg font-semibold mb-4 text-gray-900">Revenue by Project</h3>
      <div className="h-80">
        <Pie data={chartData} options={chartOptions} />
      </div>

      {/* Summary table below chart */}
      <div className="mt-6 space-y-2">
        {aggregated.map(project => {
          const total = aggregated.reduce((sum, p) => sum + p.total, 0);
          const percentage = total > 0 ? ((project.total / total) * 100).toFixed(1) : '0';
          return (
            <div key={project.id} className="flex justify-between text-sm">
              <span className="text-gray-700">{project.name}</span>
              <span className="font-medium text-gray-900">
                {formatCurrency(project.total)} ({percentage}%)
              </span>
            </div>
          );
        })}
      </div>

      {/* Grand total */}
      <div className="mt-4 pt-4 border-t border-gray-200 flex justify-between font-semibold">
        <span>Total</span>
        <span>{formatCurrency(aggregated.reduce((sum, p) => sum + p.total, 0))}</span>
      </div>
    </div>
  );
};
```

```typescript
'use client';
import React, { useState } from 'react';
import { toast } from 'react-hot-toast';

interface User {
  id: string;
  email: string;
  name: string;
  roles: string[];
  createdAt: string;
  updatedAt: string;
}

export default function ProfilePage() {
  const [user, setUser] = useState<User | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [isEditing, setIsEditing] = useState(false);
  const [editName, setEditName] = useState('');

  React.useEffect(() => {
    const fetchUser = async () => {
      try {
        const response = await fetch('/api/users/me');
        if (!response.ok) throw new Error('Failed to load profile');
        const data = await response.json();
        setUser(data);
        setEditName(data.name || '');
      } catch (err) {
        toast.error('Failed to load profile');
      } finally {
        setIsLoading(false);
      }
    };

    fetchUser();
  }, []);

  const handleSave = async () => {
    if (!user) return;
    try {
      const response = await fetch('/api/users/me', {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name: editName }),
      });
      if (!response.ok) throw new Error();
      setUser({ ...user, name: editName });
      setIsEditing(false);
      toast.success('Profile updated');
    } catch {
      toast.error('Failed to update profile');
    }
  };

  if (isLoading) {
    return (
      <div className="min-h-screen bg-gray-50 p-6 flex items-center justify-center">
        <div className="animate-spin text-gray-400">Loading...</div>
      </div>
    );
  }

  if (!user) {
    return (
      <div className="min-h-screen bg-gray-50 p-6 flex items-center justify-center">
        <div className="text-red-600">Profile not available. Please log in.</div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-gray-50 p-6">
      <div className="max-w-2xl mx-auto">
        <div className="bg-white rounded-lg border border-gray-100 p-8">
          <h1 className="text-3xl font-bold mb-8 text-gray-900">My Profile</h1>

          {/* Name */}
          <div className="mb-6">
            <label className="block text-sm font-medium text-gray-700 mb-2">Name</label>
            {isEditing ? (
              <input
                type="text"
                value={editName}
                onChange={(e) => setEditName(e.target.value)}
                className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500"
              />
            ) : (
              <div className="text-lg text-gray-900">{user.name || 'Not set'}</div>
            )}
          </div>

          {/* Email (read-only) */}
          <div className="mb-6">
            <label className="block text-sm font-medium text-gray-700 mb-2">Email</label>
            <div className="text-lg text-gray-900">{user.email}</div>
            <div className="text-xs text-gray-500 mt-1">Cannot change email</div>
          </div>

          {/* Roles */}
          <div className="mb-6">
            <label className="block text-sm font-medium text-gray-700 mb-2">Roles</label>
            <div className="flex gap-2">
              {user.roles.map(role => (
                <span key={role} className="px-3 py-1 bg-blue-100 text-blue-800 rounded-full text-sm font-medium">
                  {role}
                </span>
              ))}
            </div>
          </div>

          {/* Account dates */}
          <div className="grid grid-cols-2 gap-6 mb-8 pb-8 border-b border-gray-200">
            <div>
              <label className="text-xs font-semibold text-gray-500 uppercase">Member Since</label>
              <div className="text-gray-900">{new Date(user.createdAt).toLocaleDateString()}</div>
            </div>
            <div>
              <label className="text-xs font-semibold text-gray-500 uppercase">Last Updated</label>
              <div className="text-gray-900">{new Date(user.updatedAt).toLocaleDateString()}</div>
            </div>
          </div>

          {/* Action buttons */}
          <div className="flex gap-3">
            {isEditing ? (
              <>
                <button
                  onClick={handleSave}
                  className="px-6 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 font-medium"
                >
                  Save Changes
                </button>
                <button
                  onClick={() => setIsEditing(false)}
                  className="px-6 py-2 bg-gray-200 text-gray-900 rounded-lg hover:bg-gray-300 font-medium"
                >
                  Cancel
                </button>
              </>
            ) : (
              <button
                onClick={() => setIsEditing(true)}
                className="px-6 py-2 bg-gray-200 text-gray-900 rounded-lg hover:bg-gray-300 font-medium"
              >
                Edit Profile
              </button>
            )}
          </div>
        </div>

        {/* Additional info section */}
        <div className="mt-8 bg-white rounded-lg border border-gray-100 p-6">
          <h2 className="text-lg font-semibold mb-4 text-gray-900">Account Security</h2>
          <div className="space-y-3 text-sm">
            <div className="flex justify-between">
              <span className="text-gray-700">Two-Factor Authentication</span>
              <span className="text-gray-500">Not enabled</span>
            </div>
            <div className="flex justify-between">
              <span className="text-gray-700">Login History</span>
              <a href="#" className="text-blue-600 hover:underline">View</a>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}
```
