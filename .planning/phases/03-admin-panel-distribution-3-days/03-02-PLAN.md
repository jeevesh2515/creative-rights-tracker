   ```typescript
     e.preventDefault();
     if (!validate()) return;
     
     setIsSubmitting(true);
     try {
       const payload: DistributionPayload = {
         project_id: selectedProject,
         recipients: recipients.map(r => ({
           user_id: r.user_id,
           percentage: r.percentage,
           amount: (totalAmount * r.percentage) / 100
         })),
         total_amount: totalAmount,
         initiated_by: user.id,
         timestamp: new Date().toISOString()
       };

       const res = await fetch('/api/admin/distribute', {
         method: 'POST',
         headers: { 'Content-Type': 'application/json' },
         body: JSON.stringify(payload)
       });

       if (!res.ok) {
         const error = await res.json();
         toast.error(error.message || 'Distribution failed');
         return;
       }

       const result = await res.json();
       toast.success(`Distribution created: ${result.txHash}`);
       // Reset form
       setRecipients([{ user_id: '', percentage: 0 }]);
       setTotalAmount(0);
       setShowPreview(false);
     } catch (err) {
       toast.error(err.message);
     } finally {
       setIsSubmitting(false);
     }
   };
   ```

   ```typescript
   const provider = new ethers.JsonRpcProvider(process.env.NEXT_PUBLIC_RPC_URL);
   const signer = new ethers.Wallet(process.env.ADMIN_PRIVATE_KEY, provider);
   const contract = new ethers.Contract(
     process.env.NEXT_PUBLIC_REVENUE_RIGHTS_ADDRESS,
     REVENUE_RIGHTS_ABI,
     signer
   );

   const recipients = payload.recipients.map(r => r.user_id);
   const percentages = payload.recipients.map(r => BigInt(r.percentage * 100)); // Store as basis points

   let txHash: string;
   try {
     const tx = await contract.createDistribution(
       payload.project_id,
       recipients,
       percentages,
       {
         value: ethers.parseEther(payload.total_amount.toString())
       }
     );
     txHash = tx.hash;
     await tx.wait(); // Wait for confirmation
   } catch (contractError) {
     throw new Error(`Smart contract execution failed: ${contractError.message}`);
   }
   ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-07 | Spoofing | /api/admin/distribute | Mitigate | Verify admin role on server-side (not just client); check session token |
| T-03-08 | Tampering | Distribution Payload | Mitigate | Validate percentages sum to 100 (server-side, not just client); amounts immutable after blockchain submission |
| T-03-09 | Repudiation | Audit Log | Mitigate | Every distribution must create audit log with admin user_id, timestamp, recipients, amount |
